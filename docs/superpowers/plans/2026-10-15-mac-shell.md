```swift
import Foundation

/// 网页发来的操作请求（设计文档 5.3）
public enum OfficeAction: Equatable, Sendable {
    case ready
    case activate(String)
    case launch(String)
    case quit(String)

    /// body 是 WKScriptMessage.body（字典）或 JSON 字符串；认不出来返回 nil
    public static func parse(_ body: Any) -> OfficeAction? {
        var dict = body as? [String: Any]
        if dict == nil, let text = body as? String {
            dict = (try? JSONSerialization.jsonObject(with: Data(text.utf8))) as? [String: Any]
        }
        guard let dict, let action = dict["action"] as? String else { return nil }
        if action == "ready" { return .ready }
        guard let id = dict["id"] as? String, !id.isEmpty else { return nil }
        switch action {
        case "activate": return .activate(id)
        case "launch": return .launch(id)
        case "quit": return .quit(id)
        default: return nil
        }
    }
}

/// roster 消息里的一个员工（字段名即协议字段名）
public struct RosterItem: Encodable, Equatable, Sendable {
    public var id: String
    public var name: String
    public var kind: EmployeeKind
    public var seat: Int?
    public var color: String
    /// 64×64 PNG 的 base64，不带 data: 前缀
    public var iconPNG: String

    public init(id: String, name: String, kind: EmployeeKind, seat: Int?, color: String, iconPNG: String) {
        self.id = id
        self.name = name
        self.kind = kind
        self.seat = seat
        self.color = color
        self.iconPNG = iconPNG
    }

    private enum CodingKeys: String, CodingKey { case id, name, kind, seat, color, iconPNG }

    // 临时工的 seat 要显式写 null
    public func encode(to encoder: Encoder) throws {
        var c = encoder.container(keyedBy: CodingKeys.self)
        try c.encode(id, forKey: .id)
        try c.encode(name, forKey: .name)
        try c.encode(kind, forKey: .kind)
        try c.encode(seat, forKey: .seat)
        try c.encode(color, forKey: .color)
        try c.encode(iconPNG, forKey: .iconPNG)
    }
}

/// 推给网页的消息
public enum OutboundMessage: Equatable, Sendable {
    case roster([RosterItem])
    case status(NetworkState, [EmployeeStatus])
    case paused(Bool)

    public func json() -> String {
        let encoder = JSONEncoder()
        encoder.outputFormatting = [.sortedKeys]
        let data: Data
        switch self {
        case .roster(let employees):
            data = try! encoder.encode(RosterBody(type: "roster", employees: employees))
        case .status(let network, let employees):
            data = try! encoder.encode(StatusBody(type: "status", network: network, employees: employees))
        case .paused(let paused):
            data = try! encoder.encode(PausedBody(type: "paused", paused: paused))
        }
        return String(decoding: data, as: UTF8.self)
    }
}

private struct RosterBody: Encodable { let type: String; let employees: [RosterItem] }
private struct StatusBody: Encodable { let type: String; let network: NetworkState; let employees: [EmployeeStatus] }
private struct PausedBody: Encodable { let type: String; let paused: Bool }
```

### Task 1: OfficeFeed

决定给网页发什么（网页那边的约定）：
- 网页 `ready` 之前什么都不发，只记下最新快照。
- `ready`（每次页面加载都会来）：发完整 roster → 重置 `StatusDiffer` 后发完整 status → 发当前 paused。
- 之后每拍：名单（含图标、颜色）变了才发 roster，**roster 总在 status 前面**；status 只发变化的员工。
- `setPaused` 只在值变化且已 ready 时发。
- 调试模拟：`simulate(id, state)` 强制某员工的状态，`nil` 取消。
- 网页进程崩溃要重新加载：`reset()` 后等下一个 `ready`。
- Dock 读失败（返回空）时沿用上一次的 Dock 列表，避免所有正式员工"下班走人"。

```swift
import Foundation

public struct AppIcon: Equatable, Sendable {
    /// 64×64 PNG 的 base64
    public var pngBase64: String
    /// 主色 #RRGGBB
    public var color: String

    public init(pngBase64: String, color: String) {
        self.pngBase64 = pngBase64
        self.color = color
    }
}

public protocol IconProviding {
    func icon(for entry: RosterEntry) -> AppIcon
}

/// 决定什么时候给网页发什么：ready 之前不发；名单变了才发 roster（总在 status 前面）；状态只发变化的
public struct OfficeFeed {
    private let icons: IconProviding
    private var isReady = false
    private var paused = false
    private var last: OfficeSnapshot?
    private var sentRoster: [RosterItem]?
    private var differ = StatusDiffer()
    private var overrides: [String: EmployeeState] = [:]

    public init(icons: IconProviding) {
        self.icons = icons
    }

    /// 调试模拟中的员工
    public var simulated: [String: EmployeeState] { overrides }

    /// 每拍调一次
    public mutating func update(_ snapshot: OfficeSnapshot) -> [OutboundMessage] {
        last = snapshot
        return isReady ? messages(for: snapshot) : []
    }

    /// 网页（重新）加载完成：全量 roster → 全量 status → paused
    public mutating func ready() -> [OutboundMessage] {
        isReady = true
        sentRoster = nil
        differ.reset()
        var out = last.map { messages(for: $0) } ?? []
        out.append(.paused(paused))
        return out
    }

    public mutating func setPaused(_ value: Bool) -> [OutboundMessage] {
        guard value != paused else { return [] }
        paused = value
        return isReady ? [.paused(value)] : []
    }

    /// 调试：强制某个员工进入某个状态；nil 取消
    public mutating func simulate(_ id: String, _ state: EmployeeState?) -> [OutboundMessage] {
        overrides[id] = state
        guard isReady, let last else { return [] }
        return messages(for: last)
    }

    /// 网页进程挂了要重新加载：等下一个 ready
    public mutating func reset() {
        isReady = false
    }

    private mutating func messages(for snapshot: OfficeSnapshot) -> [OutboundMessage] {
        var out: [OutboundMessage] = []
        let roster = snapshot.roster.map { e -> RosterItem in
            let icon = icons.icon(for: e)
            return RosterItem(id: e.id, name: e.name, kind: e.kind, seat: e.seat, color: icon.color, iconPNG: icon.pngBase64)
        }
        if roster != sentRoster {
            out.append(.roster(roster))
            sentRoster = roster
        }
        let statuses = snapshot.statuses.map { s -> EmployeeStatus in
            var s = s
            if let forced = overrides[s.id] { s.state = forced }
            return s
        }
        let changed = differ.changes(statuses)
        if !changed.isEmpty { out.append(.status(snapshot.network, changed)) }
        return out
    }
}
```

```swift
import AppKit
import UniformTypeIdentifiers

public enum IconRenderer {
    /// 把 NSImage 画成 size × size 的 sRGB 位图
    public static func cgImage(_ image: NSImage, size: Int) -> CGImage? {
        guard let space = CGColorSpace(name: CGColorSpace.sRGB),
              let ctx = CGContext(data: nil, width: size, height: size, bitsPerComponent: 8, bytesPerRow: 0,
                                  space: space, bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue) else { return nil }
        NSGraphicsContext.saveGraphicsState()
        NSGraphicsContext.current = NSGraphicsContext(cgContext: ctx, flipped: false)
        image.draw(in: NSRect(x: 0, y: 0, width: size, height: size), from: .zero, operation: .copy, fraction: 1)
        NSGraphicsContext.restoreGraphicsState()
        return ctx.makeImage()
```

```swift
import Foundation

public enum AppOperation: Equatable, Sendable {
    case launch(URL)
    case activate(Int32)
    case quit(Int32)
    case none
}

public enum ActionPlanner {
    /// 网页看到的状态可能慢一拍：launch 在跑的 app 就切过去，activate 没在跑的就启动；访达不退出
    public static func plan(_ action: OfficeAction, running: [AppProcess], urlFor: (String) -> URL?) -> AppOperation {
        switch action {
        case .ready:
            return .none
        case .activate(let id), .launch(let id):
            if let p = running.first(where: { $0.bundleID == id }) { return .activate(p.pid) }
            return urlFor(id).map(AppOperation.launch) ?? .none
        case .quit(let id):
            guard id != Roster.finderID, let p = running.first(where: { $0.bundleID == id }) else { return .none }
            return .quit(p.pid)
        }
    }
}
```

```swift
import Foundation

/// office://app/... → 网页资源目录里的文件
public enum WebAssets {
    public static let scheme = "office"
    public static let indexURL = URL(string: "office://app/index.html")!

    static let mimeTypes = [
        "html": "text/html", "js": "text/javascript", "mjs": "text/javascript", "css": "text/css",
        "json": "application/json", "png": "image/png", "svg": "image/svg+xml", "ico": "image/x-icon",
        "woff2": "font/woff2",
    ]

    /// URL 路径 → 文件和 MIME；不存在、是目录、跑出资源目录都返回 nil
    public static func resolve(root: URL, path: String) -> (url: URL, mime: String)? {
        var relative = path.removingPercentEncoding ?? path
        while relative.hasPrefix("/") { relative.removeFirst() }
        if relative.isEmpty { relative = "index.html" }
        let base = root.standardizedFileURL
        let file = base.appendingPathComponent(relative).standardizedFileURL
        let prefix = base.path.hasSuffix("/") ? base.path : base.path + "/"
        var isDirectory: ObjCBool = false
        guard file.path.hasPrefix(prefix),
              FileManager.default.fileExists(atPath: file.path, isDirectory: &isDirectory),
              !isDirectory.boolValue else { return nil }
        return (file, mimeTypes[file.pathExtension.lowercased()] ?? "application/octet-stream")
    }

    /// 网页资源目录：APP_OFFICE_WEB → app 包里的 Resources/web → 仓库里的 web/dist（开发时）
    public static func locateRoot(bundle: Bundle = .main,
                                  environment: [String: String] = ProcessInfo.processInfo.environment,
                                  sourceFile: String = #filePath) -> URL? {
        let hasIndex = { (dir: URL) in FileManager.default.fileExists(atPath: dir.appendingPathComponent("index.html").path) }
        if let custom = environment["APP_OFFICE_WEB"] {
            let dir = URL(fileURLWithPath: custom)
            if hasIndex(dir) { return dir }
        }
        if let bundled = bundle.resourceURL?.appendingPathComponent("web"), hasIndex(bundled) { return bundled }
        // sourceFile 是 <仓库>/Sources/<target>/<文件>.swift
        let repo = URL(fileURLWithPath: sourceFile).deletingLastPathComponent().deletingLastPathComponent().deletingLastPathComponent()
        let dist = repo.appendingPathComponent("web/dist")
        return hasIndex(dist) ? dist : nil
    }
}
```

```swift
final class WebAssetHandler: NSObject, WKURLSchemeHandler {
    private let root: URL

    init(root: URL) {
        self.root = root
    }

    func webView(_ webView: WKWebView, start task: WKURLSchemeTask) {

              let hit = WebAssets.resolve(root: root, path: url.path),

        task.didFinish()
    }

    func webView(_ webView: WKWebView, stop task: WKURLSchemeTask) {}
}

/// 网页 ↔ Swift：发消息给 window.office.receive，收 webkit.messageHandlers.office 的操作
final class WebBridge: NSObject, WKScriptMessageHandler, WKNavigationDelegate {
    let webView: OfficeWebView
    var onAction: ((OfficeAction) -> Void)?
    /// 网页进程挂了（会自动重新加载）
    var onCrash: (() -> Void)?
    var onLoadFailed: ((String) -> Void)?
    private let logging = ProcessInfo.processInfo.environment["APP_OFFICE_LOG"] != nil

    init(webRoot: URL) {
        let config = WKWebViewConfiguration()
        config.setURLSchemeHandler(WebAssetHandler(root: webRoot), forURLScheme: WebAssets.scheme)
        webView = OfficeWebView(frame: .zero, configuration: config)
        super.init()
        // 必须在加载页面之前注册：网页启动时只检查一次，没有就进 mock 模式
        webView.configuration.userContentController.add(WeakMessageHandler(self), name: "office")
        webView.navigationDelegate = self
        webView.isInspectable = true // Safari 开发菜单里能调试
    }

    func load() {
        webView.load(URLRequest(url: WebAssets.indexURL))
    }

    func send(_ messages: [OutboundMessage]) {
        for message in messages {
            let json = message.json()
            if logging { print("→ web", json.prefix(160)) }
            webView.callAsyncJavaScript("window.office.receive(m); return 0;", arguments: ["m": json],
                                        in: nil, in: .page) { [logging] result in
                if logging, case .failure(let error) = result { print("send failed:", error.localizedDescription) }
            }
        }
    }

    func userContentController(_ controller: WKUserContentController, didReceive message: WKScriptMessage) {
        guard let action = OfficeAction.parse(message.body) else { return }
        if logging { print("← web", action) }
        onAction?(action)
    }

    func webViewWebContentProcessDidTerminate(_ webView: WKWebView) {
        onCrash?()
        load()
    }

    func webView(_ webView: WKWebView, didFail navigation: WKNavigation!, withError error: Error) {
        onLoadFailed?(error.localizedDescription)
    }

    func webView(_ webView: WKWebView, didFailProvisionalNavigation navigation: WKNavigation!, withError error: Error) {
        onLoadFailed?(error.localizedDescription)
    }
}

/// WKUserContentController 会强引用 handler，转一层弱引用避免循环引用
private final class WeakMessageHandler: NSObject, WKScriptMessageHandler {
    weak var target: WKScriptMessageHandler?

    init(_ target: WKScriptMessageHandler) {
        self.target = target
    }

    func userContentController(_ controller: WKUserContentController, didReceive message: WKScriptMessage) {
        target?.userContentController(controller, didReceive: message)
    }
}
```

```swift
import AppKit
import OfficeCore

/// 每秒一拍：主线程收集运行列表和 Dock，后台串行队列算状态，回主线程交给 onSnapshot
final class MonitorLoop {
    var onSnapshot: ((OfficeSnapshot) -> Void)?
    /// 最近一拍的运行列表（执行网页操作时用）
    private(set) var running: [AppProcess] = []
    private let network = NetworkWatcher()
    private let monitor: OfficeMonitor
    private let workspace = WorkspaceAppsSource()
    private let worker = DispatchQueue(label: "AppOffice.tick")
    private var dock: [DockApp] = []
    private var ticks = 0
    private var busy = false
    private var timer: Timer?

    init() {
        monitor = OfficeMonitor(network: network)
    }

    func start() {
        step()
        let timer = Timer(timeInterval: 1, repeats: true) { [weak self] _ in self?.step() }
        RunLoop.main.add(timer, forMode: .common) // 打开菜单时也不停
        self.timer = timer
    }

    private func step() {
        // 上一拍还没算完就跳过，避免排队
        guard !busy else { return }
        if ticks % 10 == 0 { dock = DockReader.keepLastIfEmpty(DockReader.load(), previous: dock) }
        ticks += 1
        busy = true
        let currentDock = dock
        let currentRunning = workspace.runningApps()
        running = currentRunning
        worker.async { [weak self] in
            guard let self else { return }
            self.network.refreshTraffic()
            let snapshot = self.monitor.tick(dock: currentDock, running: currentRunning)
            DispatchQueue.main.async {
                self.busy = false
                self.onSnapshot?(snapshot)
            }
        }
    }
}
```

```swift
/// 执行网页发来的启动 / 切换 / 退出
enum AppControl {
    static func perform(_ operation: AppOperation, onLaunchFailed: @escaping (String) -> Void) {
        switch operation {
        case .none:
            break
        case .activate(let pid):
            NSRunningApplication(processIdentifier: pid)?.activate()
        case .quit(let pid):
            // 等同 ⌘Q：有没保存的文档时 app 自己会弹窗
            NSRunningApplication(processIdentifier: pid)?.terminate()
        case .launch(let url):
            let config = NSWorkspace.OpenConfiguration()
            config.activates = true
            NSWorkspace.shared.openApplication(at: url, configuration: config) { _, error in
                guard let error else { return }
                let name = url.deletingPathExtension().lastPathComponent
                DispatchQueue.main.async { onLaunchFailed("\(name) 没能打开：\(error.localizedDescription)") }
            }
        }
    }
}
```

```swift
    private let toggles: [Toggle]

    private let enableHangDetection: () -> Void
    private let employees: () -> [(id: String, name: String)]

         hangDetectionAvailable: @escaping () -> Bool,
         enableHangDetection: @escaping () -> Void,

         simulated: @escaping () -> [String: EmployeeState],

            let entry = NSMenuItem(title: toggle.title, action: #selector(flip(_:)), keyEquivalent: "")
            entry.target = self
            entry.tag = index
            entry.state = toggle.get() ? .on : .off
            menu.addItem(entry)
        }
        menu.addItem(.separator())
        if !hangDetectionAvailable() {
            let entry = NSMenuItem(title: "开启卡顿检测…", action: #selector(askForAccessibility), keyEquivalent: "")
            entry.target = self
            menu.addItem(entry)
        }
        let debug = NSMenuItem(title: "调试", action: nil, keyEquivalent: "")
        let debugMenu = NSMenu()
        let simulateItem = NSMenuItem(title: "模拟状态", action: nil, keyEquivalent: "")
        simulateItem.submenu = simulationMenu()
        debugMenu.addItem(simulateItem)
        debug.submenu = debugMenu
        menu.addItem(debug)
        menu.addItem(.separator())
        menu.addItem(NSMenuItem(title: "退出 AppOffice", action: #selector(NSApplication.terminate(_:)), keyEquivalent: "q"))
    }

    private func simulationMenu() -> NSMenu {
        let menu = NSMenu()
        let current = simulated()
        for employee in employees() {
            let forced = current[employee.id]
            let person = NSMenuItem(title: forced == nil ? employee.name : "\(employee.name)（模拟中）", action: nil, keyEquivalent: "")
            let states = NSMenu()
            for (state, title) in Self.stateTitles {
                let entry = NSMenuItem(title: title, action: #selector(pickState(_:)), keyEquivalent: "")
                entry.target = self
                entry.representedObject = [employee.id, state.rawValue]
                entry.state = forced == state ? .on : .off
                states.addItem(entry)

            states.addItem(.separator())
            let clear = NSMenuItem(title: "取消模拟", action: #selector(pickState(_:)), keyEquivalent: "")

            clear.representedObject = [employee.id, ""]

            menu.addItem(person)

        menu.addItem(clearAll)

        toggle.set(!toggle.get())

    @objc private func askForAccessibility() {
        enableHangDetection()
    }

    @objc private func pickState(_ sender: NSMenuItem) {
        guard let pair = sender.representedObject as? [String], pair.count == 2 else { return }
        simulate(pair[0], EmployeeState(rawValue: pair[1]))
    }

    @objc private func clearAllSimulations() {
        for id in simulated().keys { simulate(id, nil) }
    }
```

```swift
import AppKit
import ApplicationServices
import OfficeCore
import UserNotifications

final class AppDelegate: NSObject, NSApplicationDelegate {
    private let defaults = UserDefaults.standard
    private let loop = MonitorLoop()
    private var feed = OfficeFeed(icons: WorkspaceIcons())
    private var latest: OfficeSnapshot?
    private var bridge: WebBridge?
    private var panel: OfficePanel!
    private var menuBar: MenuBarController!
    private let notice = NSTextField(labelWithString: "")
    private var hovering = false
    private var screenLocked = false

    func applicationDidFinishLaunching(_ notification: Notification) {
        defaults.register(defaults: ["showOffice": true, "pinOnTop": true, "translucent": false])
        panel = OfficePanel(content: makeContent())
        panel.pinned = defaults.bool(forKey: "pinOnTop")
        panel.onHoverChange = { [weak self] inside in
            self?.hovering = inside
            self?.applyTranslucency()
        }
        observeVisibility()
        menuBar = makeMenuBar()

        loop.onSnapshot = { [weak self] snapshot in
            guard let self else { return }
            self.latest = snapshot
            self.bridge?.send(self.feed.update(snapshot))
        }
        loop.start()
        bridge?.load()
        if defaults.bool(forKey: "showOffice") { panel.orderFrontRegardless() }
        applyTranslucency()
        updatePaused()
    }

    // MARK: - 窗口内容

    private func makeContent() -> NSView {
        let container = NSView()
        notice.alignment = .center
        notice.textColor = .secondaryLabelColor
        notice.isHidden = true
        notice.autoresizingMask = [.width, .minYMargin, .maxYMargin]
        if let root = WebAssets.locateRoot() {
            let bridge = WebBridge(webRoot: root)
            bridge.webView.autoresizingMask = [.width, .height]
            container.addSubview(bridge.webView)
            bridge.onAction = { [weak self] action in self?.handle(action) }
            bridge.onCrash = { [weak self] in self?.feed.reset() }
            bridge.onLoadFailed = { [weak self] message in self?.showNotice("网页没加载出来：\(message)") }
            self.bridge = bridge
        } else {
            showNotice("找不到网页资源：先在 web/ 里运行 npm run build")
        }
        container.addSubview(notice)
        container.postsFrameChangedNotifications = true
        NotificationCenter.default.addObserver(forName: NSView.frameDidChangeNotification, object: container, queue: .main) { [weak self] _ in
            self?.layout(container)
        }
        return container
    }

    private func layout(_ container: NSView) {
        bridge?.webView.frame = container.bounds
        notice.frame = NSRect(x: 12, y: container.bounds.midY - 10, width: max(0, container.bounds.width - 24), height: 20)
    }

    private func showNotice(_ text: String) {
        notice.stringValue = text
        notice.isHidden = false
    }

    // MARK: - 网页操作

    private func handle(_ action: OfficeAction) {
        if action == .ready {
            notice.isHidden = true
            bridge?.send(feed.ready())
            updatePaused()
            return
        }
        let operation = ActionPlanner.plan(action, running: loop.running) {
            NSWorkspace.shared.urlForApplication(withBundleIdentifier: $0)
        }
        AppControl.perform(operation) { [weak self] message in self?.notifyLaunchFailed(message) }
    }

    private func notifyLaunchFailed(_ message: String) {
        NSLog("AppOffice: %@", message)
        // swift run 时没有 bundle id，用不了通知中心
        guard Bundle.main.bundleIdentifier != nil else { return }
        let center = UNUserNotificationCenter.current()
        center.requestAuthorization(options: [.alert]) { granted, _ in
            guard granted else { return }
            let content = UNMutableNotificationContent()
            content.title = "AppOffice"
            content.body = message
            center.add(UNNotificationRequest(identifier: UUID().uuidString, content: content, trigger: nil))
        }
    }

    // MARK: - 显示、钉住、透明、暂停

    private var showOffice: Bool {
        get { panel.isVisible }
        set {
            defaults.set(newValue, forKey: "showOffice")
            if newValue { panel.orderFrontRegardless() } else { panel.orderOut(nil) }
            updatePaused()
        }
    }

    private func applyTranslucency() {
        panel.alphaValue = defaults.bool(forKey: "translucent") && !hovering ? 0.5 : 1
    }

    private func observeVisibility() {
        let center = NotificationCenter.default
        center.addObserver(self, selector: #selector(visibilityChanged), name: NSWindow.didChangeOcclusionStateNotification, object: panel)
        center.addObserver(self, selector: #selector(panelWillClose), name: NSWindow.willCloseNotification, object: panel)
        let distributed = DistributedNotificationCenter.default()
        distributed.addObserver(self, selector: #selector(screenDidLock), name: .init("com.apple.screenIsLocked"), object: nil)
        distributed.addObserver(self, selector: #selector(screenDidUnlock), name: .init("com.apple.screenIsUnlocked"), object: nil)
    }

    @objc private func visibilityChanged() { updatePaused() }

    @objc private func panelWillClose() {
        defaults.set(false, forKey: "showOffice")
        DispatchQueue.main.async { self.updatePaused() }
    }

    @objc private func screenDidLock() {
        screenLocked = true
        updatePaused()
    }

    @objc private func screenDidUnlock() {
        screenLocked = false
        updatePaused()
    }

    /// 窗口隐藏、被完全挡住或锁屏时网页停止渲染
    private func updatePaused() {
        let paused = !panel.isVisible || !panel.occlusionState.contains(.visible) || screenLocked
        bridge?.send(feed.setPaused(paused))
    }

    // MARK: - 菜单栏

    private func makeMenuBar() -> MenuBarController {
        MenuBarController(
            toggles: [
                .init(title: "显示办公室", get: { [weak self] in self?.showOffice ?? false },
                      set: { [weak self] in self?.showOffice = $0 }),
                .init(title: "钉在最上层", get: { [weak self] in self?.panel.pinned ?? false },
                      set: { [weak self] in
                          self?.defaults.set($0, forKey: "pinOnTop")
                          self?.panel.pinned = $0
                      }),
                .init(title: "透明模式", get: { [weak self] in self?.defaults.bool(forKey: "translucent") ?? false },
                      set: { [weak self] in
                          self?.defaults.set($0, forKey: "translucent")
                          self?.applyTranslucency()
                      }),
            ],
            hangDetectionAvailable: { [weak self] in self?.latest?.hangDetectionAvailable ?? true },
            enableHangDetection: { Self.askForAccessibility() },
            employees: { [weak self] in self?.latest?.roster.map { (id: $0.id, name: $0.name) } ?? [] },
            simulated: { [weak self] in self?.feed.simulated ?? [:] },
            simulate: { [weak self] id, state in
                guard let self else { return }
                self.bridge?.send(self.feed.simulate(id, state))
            })
    }

    /// 弹系统的辅助功能授权提示，并打开系统设置对应页面
    private static func askForAccessibility() {
        let options = [kAXTrustedCheckOptionPrompt.takeUnretainedValue() as String: true] as CFDictionary
        guard !AXIsProcessTrustedWithOptions(options),
              let url = URL(string: "x-apple.systempreferences:com.apple.preference.security?Privacy_Accessibility") else { return }
        NSWorkspace.shared.open(url)
    }
}
```

## 实施记录（与上文计划的偏差）

逐任务审查和最终审查之后，实际代码在这些地方和上面的计划不同，以代码为准：

| 位置 | 改动 | 原因 |
|---|---|---|
| `WebAssets.resolve`（b3aaeae） | 约定传入的是已经解码过的 `URL.path`，不再做百分号解码 | `WKURLSchemeTask` 给的 `URL.path` 已经解码过，再解一次会把文件名里字面的 `%` 解错（`a%20b` 变成 `a b`） |
| `WorkspaceIcons`（b3aaeae） | 找不到 app 时用通用 app 图标，且不记在这个 app 名下（下次再找）；画不出来也退回通用图标，最后才给空图标 | 网页要求 `iconPNG` 必须是有效 PNG；app 稍后装上时能换成真图标 |
| `OfficeFeed`（b3aaeae） | 只有网络状态变了、没有员工状态变化时也发 `status`（员工列表为空） | 原写法断网/联网不会通知网页，没人换状态时健身房没反应 |
| `AppControl` `.activate`（a524af1） | 切换已在跑的 app 改用 `NSWorkspace.openApplication(at:)`（LaunchServices），拿不到 URL 才用 `activate()` | macOS 14+ 上 accessory app 调 `activate()` 会被系统静默拒绝 |
| `WebBridge` 日志（a524af1） | `APP_OFFICE_LOG` 日志只打摘要：roster 打人数和字节数，status 打 `id=state` | roster 带 base64 图标，原来截前 160 字符看不出内容 |
| `AppDelegate.handle`（a524af1） | 启动时优先用名单里记下的 `bundleURL`，再问 LaunchServices；两者都没有时通知"找不到这个 app：名字" | Dock 里的 app 不一定能按 bundle id 找到；原写法什么都不发生 |
| `WebBridge` 崩溃重载（a524af1） | 网页进程挂了隔 1 秒再加载；60 秒内崩 3 次就放弃并在窗口里提示 | 避免无限崩溃重载占满 CPU |
| `MonitorLoop` Dock（a524af1） | Dock 还没读到（空）时每拍都重试，读到后才回到每 10 拍一次 | 启动时偶尔读不到 Dock，原写法要等 10 秒才有正式员工 |
| `AppDelegate` 悬停（a524af1） | 显示窗口时按鼠标当前位置重新算悬停；关窗时清掉悬停 | 窗口隐藏期间收不到鼠标进出事件，透明模式会停在错的状态 |
| `MenuBarController`（a524af1） | 模拟菜单关掉 `autoenablesItems`，各项显式启用 | 否则"取消全部模拟"的 `isEnabled` 不生效 |
| `WebBridge` 加载失败（a524af1） | `NSURLErrorCancelled`（-999）不算加载失败 | 新的加载打断旧的时会报 -999，误显示"网页没加载出来" |
| `WebBridge` 导航（本次） | `decidePolicyFor navigationAction` 只放行 `office://`，其余取消；被拦掉的 WebKit 102 错误不算加载失败 | 网页里的链接、跳转不应该把办公室换成别的页面 |
| `OfficeWebView`（本次） | 重写 `draggingEntered` / `draggingUpdated` 返回空、`prepareForDragOperation` / `performDragOperation` 返回 false | 拖进来的文件会被 WKWebView 直接打开；`unregisterDraggedTypes()` 可能被 WebKit 重新注册，拦回调更稳 |
| `MonitorLoop` / `OfficeMonitor`（本次） | 网页暂停时 20 秒一拍（容差 2 秒），不跑 nettop、不做卡顿检测（`tick(detectHangs: false)`，谁都不算卡住，`hangDetectionAvailable` 沿用上次）；恢复时立刻补一拍并回到 1 秒（容差 0.2 秒） | 窗口看不见时没必要每秒 AX ping 所有 app |
| `AppControl` `.launch`（本次） | 按路径启动失败时，用 `urlForApplication(withBundleIdentifier:)` 找到的不同位置再试一次，都失败才通知 | app 被挪走、名单里的路径过期 |
| `AppDelegate.handle`（本次） | 名单里的 `bundleURL` 必须是本地文件且还在才用，否则问 LaunchServices | 同上 |
| `DockReader.parse`（本次） | `_CFURLStringType` 为 0 时按路径建 `URL(fileURLWithPath:)` | 有些 Dock 项存的是路径不是 URL（计划 1 的要求） |
| `AppDelegate` 重新打开（本次） | `applicationShouldHandleReopen` 显示办公室 | 在访达里再双击 AppOffice 时原来没反应 |
| `AppDelegate` 辅助功能（本次） | 每次都调 `AXIsProcessTrustedWithOptions(prompt: true)`；第一次只弹系统提示，之后（`askedAccessibility`）没授权就直接打开系统设置 | 原写法第一次会同时弹提示和打开系统设置；系统提示只弹一次 |
| `WorkspaceIcons`（本次） | 通用 app 图标只画一次（`static let generic`） | 找不到的 app 每拍都重画一遍图标 |
| `AppDelegate` 通知（本次） | 打包运行时设 `UNUserNotificationCenterDelegate`，`willPresent` 返回 `[.banner]` | 自己在前台时通知默认不显示 |
